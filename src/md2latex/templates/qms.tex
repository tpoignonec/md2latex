% Title page for QMS documents
\newcommand{\QMStitlepage}{%
  \begin{titlepage}%
    \thispagestyle{empty}%
    \setlength{\parindent}{0mm}%
    \setlength{\parskip}{0mm}%
    \vspace*{\stretch{1}}%
    \rule{\linewidth}{1mm}%
    \begin{flushleft}%
      {\Huge\textbf{\QMSprjname}}\\[5mm]%
      \QMSdoctype%
    \end{flushleft}%
    \rule{\linewidth}{1mm}%
    \vspace*{\stretch{1}}%
    \begin{center}%
      {\large\textbf{Revision \QMSrevision}}\\%
      $if(date)$$date$$else$\today$endif$%
    \end{center}%
    \vspace*{\stretch{1}}%
    \begin{center}%
      $if(PDcompany)$%
        {\large{}Copyright \copyright{} \QMScompany}\\%
        \QMSauthor%
      $else$%
        {\large{}Copyright \copyright{} \QMSauthor}%
      $endif$%
    \end{center}%
    \vspace*{\stretch{1}}%
    
    % QMS signature fields
    \begin{tabular}{|p{0.3\linewidth}|p{0.3\linewidth}|p{0.3\linewidth}|}%
      \hline%
      \textbf{Created:} & \textbf{Reviewed:} & \textbf{Approved:} \\%
      \hline%
      & & \\[2cm]%
      \footnotesize{name, dept., date} & \footnotesize{name, dept., date} & \footnotesize{name, dept., date} \\%
      \hline%
    \end{tabular}%
    \vspace*{\stretch{1}}%
  \end{titlepage}%
}
