\newcommand{\PDtitlepage}{%
  \begin{titlepage}%
    \thispagestyle{empty}%
    \setlength{\parindent}{0mm}%
    \setlength{\parskip}{0mm}%
    \vspace*{\stretch{1}}%
    \rule{\linewidth}{1mm}%
    \begin{flushleft}%
      {\Huge\textbf{Project ``\PDprjname{}''}}\\[5mm]%
      \PDdoctype%
    \end{flushleft}%
    \rule{\linewidth}{1mm}%
    \vspace*{\stretch{1}}%
    \begin{center}%
      {\large\textbf{Revision \PDrevision}}\\%
      $if(date)$$date$$else$\today$endif$%
    \end{center}%
    \vspace*{\stretch{1}}%
    \begin{center}%
      $if(PDcompany)$%
        {\large{}Copyright \copyright{} \PDcompany}\\%
        \PDauthor%
      $else$%
        {\large{}Copyright \copyright{} \PDauthor}%
      $endif$%
    \end{center}%
    \vspace*{\stretch{2}}%
    
    % Signature fields (can be disabled)
    $if(nosignfield)$$else$%
    \begin{tabularx}{\linewidth}{XXX}%
      \toprule%
      \textbf{Created:} & \textbf{Checked:} & \textbf{Released:} \\%
      \midrule%
      & & \\[2cm]%
      \footnotesize{name, dept., date} & \footnotesize{name, dept., date} & \footnotesize{name, dept., date} \\%
      \bottomrule%
    \end{tabularx}%
    $endif$%
